// ============================================================ // DASHBOARD DOCAS — Google Apps Script — versão final // Aba: "Carregamentos | 3P" // Cabeçalhos na linha 3 (índice 2) // ============================================================ const SHEET_ID = '1FgSbJNBg0l4GWcVYir8RZun1hNN2sugTrsl1-EVE2DY'; function doGet(e) { const output = ContentService .createTextOutput(JSON.stringify(obterDados())) .setMimeType(ContentService.MimeType.JSON); return output; } function doOptions(e) { return ContentService .createTextOutput('') .setMimeType(ContentService.MimeType.TEXT); } function obterDados() { try { const ss = SpreadsheetApp.openById(SHEET_ID); const aba = ss.getSheetByName('Carregamentos | 3P') || ss.getSheets()[0]; const dados = aba.getDataRange().getValues(); // Linha 3 = cabeçalhos (índice 2) const headers = dados[2].map(h => String(h).trim()); const col = { data: headers.indexOf('Data'), turno: headers.indexOf('Turno'), destino: headers.indexOf('Destino'), tipo: headers.indexOf('Tipo de Veículo'), cpt: headers.indexOf('CPT'), stPux: headers.indexOf('Status Puxada'), stCar: headers.indexOf('Status Carregamento'), doca: headers.indexOf('Stage Out'), // doca do veículo transp: headers.indexOf('Transportadora'), unid: headers.indexOf('Unidades'), just: headers.indexOf('Justificativo (Perda do CPT)'), }; const limite7 = new Date(); limite7.setDate(limite7.getDate() - 7); const todos = []; let dataMaisRecente = null; // Dados a partir da linha 4 (índice 3) for (let i = 3; i < dados.length; i++) { const linha = dados[i]; const celData = linha[col.data]; if (!celData || !(celData instanceof Date)) continue; const dataStr = fmt(celData); if (!dataMaisRecente || celData > dataMaisRecente) dataMaisRecente = celData; todos.push({ data: dataStr, dataObj: celData, turno: norm(linha[col.turno]), destino: norm(linha[col.destino]), tipo: norm(linha[col.tipo]), cpt: norm(linha[col.cpt]), stPux: norm(linha[col.stPux]), stCar: norm(linha[col.stCar]), doca: norm(linha[col.doca]), transp: norm(linha[col.transp]), unid: Number(linha[col.unid] || 0), just: norm(linha[col.just]), }); } // Usa a data mais recente disponível na planilha const dataAlvo = fmt(dataMaisRecente || new Date()); const registrosAlvo = todos.filter(r => r.data === dataAlvo); const registros7 = todos.filter(r => r.dataObj >= limite7); // Debug: pega primeiros 5 registros com doca preenchida const comDoca = registrosAlvo.filter(r => r.doca && r.doca !== ''); const debugDocas = comDoca.slice(0, 5).map(r => ({ doca: r.doca, stPux: r.stPux, stCar: r.stCar, status: mapStatus(r.stCar, r.stPux) })); // D+1: registros do dia seguinte const amanha = new Date(dataMaisRecente || new Date()); amanha.setDate(amanha.getDate() + 1); const amanhaStr = fmt(amanha); const registrosD1 = todos.filter(r => r.data === amanhaStr); return { ok: true, dataExibida: dataAlvo, totalRegistros: todos.length, totalComDoca: comDoca.length, debugDocas: debugDocas, carregamentos: registrosAlvo, volumePorTurno: calcVolumeTurno(registrosAlvo), statusDocas: calcStatusDocas(registrosAlvo), historicoVolume: calcHistorico(todos), perdas: calcPerdas(registros7), planoVeiculos: calcPlano(registrosAlvo), d1: { data: amanhaStr, total: registrosD1.length, volumePorTurno: calcVolumeTurno(registrosD1), completas: registrosD1.filter(r => { const c = r.stCar.toUpperCase(); return c.includes('COMPLETA') || c.includes('CARREGADA') || c.includes('ENVIADA'); }).length, }, ultimaAtualizacao: new Date().toISOString() }; } catch (err) { return { ok: false, erro: err.message }; } } function norm(v) { return String(v === null || v === undefined ? '' : v).trim(); } function fmt(d) { return Utilities.formatDate(d, 'America/Sao_Paulo', 'dd/MM/yyyy'); } function calcVolumeTurno(regs) { const plan = { T1: 0, T2: 0, T3: 0 }; const real = { T1: 0, T2: 0, T3: 0 }; const statusRealizado = ['CARREGADA', 'ENVIADA']; regs.forEach(r => { const t = r.turno.replace(/\s/g,'').toUpperCase(); let chave = null; if (t.includes('T1') || t === '1') chave = 'T1'; else if (t.includes('T2') || t === '2') chave = 'T2'; else if (t.includes('T3') || t === '3') chave = 'T3'; if (!chave) return; plan[chave]++; // planejado = todas as linhas const stCar = r.stCar.toUpperCase().trim(); if (statusRealizado.some(s => stCar.includes(s))) { real[chave]++; } }); return { T1: { planejado: plan.T1, realizado: real.T1 }, T2: { planejado: plan.T2, realizado: real.T2 }, T3: { planejado: plan.T3, realizado: real.T3 }, }; } function calcStatusDocas(regs) { const mapa = {}; const prio = { completa:5, carregando:3, aguardando:2, livre:1, outro:0 }; regs.forEach(r => { if (!r.doca) return; const st = mapStatus(r.stCar, r.stPux); if (!mapa[r.doca] || (prio[st]||0) > (prio[mapa[r.doca].status]||0)) { mapa[r.doca] = { doca: r.doca, sigla: r.destino, status: st, transp: r.transp }; } }); return Object.values(mapa); } function mapStatus(stCar, stPux) { const p = stPux.toUpperCase().trim(); const c = stCar.toUpperCase().trim(); // Ag. piso: sendo abastecida ainda if (p.includes('AG COMP') || p.includes('AGUARDANDO')) return 'aguardando'; // Livre: puxada finalizada + já carregada ou enviada (doca liberada) if (p.includes('FINALIZADO') && (c.includes('CARREGADA') || c.includes('ENVIADA'))) return 'livre'; // Carregando: puxada finalizada + caminhão carregando agora if (p.includes('FINALIZADO') && c.includes('CARREGANDO')) return 'carregando'; // Completa: puxada finalizada + pronta para carregar (sem status de carregamento ou completa) if (p.includes('FINALIZADO') && (c === '' || c.includes('COMPLETA') || c.includes('POSTERGADO'))) return 'completa'; // Bloqueada / Insumo if (c.includes('BLOQUEADA')) return 'bloqueada'; if (c.includes('INSUMO')) return 'insumo'; // Sem dados if (!p && !c) return 'livre'; return 'outro'; } function calcHistorico(regs) { const cont = {}; regs.forEach(r => { cont[r.data] = (cont[r.data] || 0) + 1; }); return Object.entries(cont) .sort(([a],[b]) => { const n = s => s.split('/').reverse().join(''); return n(a).localeCompare(n(b)); }) .slice(-15) // últimos 15 dias com dados .map(([data, total]) => ({ data, total })); } function calcPerdas(regs) { const gatilhos = ['FALTA DE PISO','INFRUTÍFERO','NOSHOW','CANCELADO','PERDIDO','AG COMP']; return regs.filter(r => { const s = (r.stCar + ' ' + r.stPux + ' ' + r.just).toUpperCase(); return gatilhos.some(g => s.includes(g)); }).map(r => ({ data: r.data, turno: r.turno, destino: r.destino, status: r.stCar || r.stPux, motivo: r.just || r.stPux || r.stCar, })); } function calcPlano(regs) { const cont = {}; regs.forEach(r => { if (!r.destino) return; cont[r.destino] = (cont[r.destino] || 0) + 1; }); return Object.entries(cont) .map(([destino, veiculos]) => ({ destino, veiculos })) .sort((a,b) => b.veiculos - a.veiculos); }
